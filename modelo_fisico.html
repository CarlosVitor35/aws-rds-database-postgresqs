    Empresa {
        INTEGER ID_Empresa PK
        VARCHAR Razao_Social
        VARCHAR Nome_Fantasia
        VARCHAR CNPJ UK
        DATE Data_Cadastro
        VARCHAR Status
    }

    Filial {
        INTEGER ID_Filial PK
        INTEGER ID_Empresa FK
        VARCHAR Razao_Social
        VARCHAR Nome_Fantasia
        VARCHAR CNPJ UK
        DATE Data_Abertura
        VARCHAR Status
    }

    Cargo {
        INTEGER ID_Cargo PK
        VARCHAR Nome_Cargo
        DECIMAL Salario
    }

    Funcionario {
        INTEGER ID_Funcionario PK
        INTEGER ID_Filial FK
        INTEGER ID_Cargo FK
        VARCHAR Nome
        VARCHAR Sobrenome
        DATE Data_Nascimento
        VARCHAR CPF UK
        VARCHAR E_mail UK
        VARCHAR Celular UK
        VARCHAR Celular_2
        DATE Data_Admissao
        VARCHAR Status
    }

    Cliente {
        INTEGER ID_Cliente PK
        VARCHAR Nome
        VARCHAR Sobrenome
        DATE Data_Nascimento
        VARCHAR CPF UK
        VARCHAR Celular UK
        VARCHAR Celular_2
        VARCHAR E_mail UK
        DATE Data_Cadastro
        VARCHAR Status
    }

    Endereco {
        INTEGER ID_Endereco PK
        INTEGER ID_Cliente FK
        INTEGER ID_Filial FK
        INTEGER ID_Empresa FK
        VARCHAR Logradouro
        VARCHAR Numero
        VARCHAR Complemento
        VARCHAR Bairro
        VARCHAR Cidade
        VARCHAR Estado
        VARCHAR CEP
    }

    Categoria {
        INTEGER ID_Categoria PK
        VARCHAR Nome UK
        VARCHAR Descricao
        VARCHAR Status
        DATE Data_Cadastro
    }

    Estoque {
        INTEGER ID_Estoque PK
        INTEGER Quantidade
        INTEGER Estoque_Minimo
    }

    Produto {
        INTEGER ID_Produto PK
        INTEGER ID_Categoria FK
        INTEGER ID_Estoque FK
        VARCHAR Nome
        VARCHAR Descricao
        DECIMAL Preco
        DATE Data_Cadastro
        VARCHAR Status
        VARCHAR Codigo_Barras UK
    }

    Fornecedor {
        INTEGER ID_Fornecedor PK
        VARCHAR Razao_Social
        VARCHAR Nome_Fantasia
        VARCHAR CNPJ UK
        VARCHAR Celular UK
        VARCHAR Celular_2
        VARCHAR E_mail UK
        DATE Data_Cadastro
        VARCHAR Status
    }

    Fornecedor_Produto {
        INTEGER ID_Fornecedor PK_FK
        INTEGER ID_Produto PK_FK
    }

    Pagamento {
        INTEGER ID_Pagamento PK
        VARCHAR Forma_Pagamento
        DECIMAL Valor
        DATE Data_Pagamento
        VARCHAR Status
    }

    Pedido {
        INTEGER ID_Pedido PK
        INTEGER ID_Endereco FK
        INTEGER ID_Cliente FK
        INTEGER ID_Funcionario FK
        INTEGER ID_Pagamento FK
        INTEGER ID_Filial FK
        DECIMAL Valor_Total
        VARCHAR Status
        DATE Data_Pedido
        VARCHAR Observacao
    }

    Itens_Pedido {
        INTEGER ID_Item PK
        INTEGER ID_Pedido FK
        INTEGER ID_Produto FK
        INTEGER Quantidade
        DECIMAL Preco_Unico
    }

    Movimentacao_Estoque {
        INTEGER ID_Movimentacao PK
        INTEGER ID_Estoque FK
        INTEGER ID_Produto FK
        VARCHAR Tipo
        INTEGER Quantidade
        DATE Data_Movimentacao
        VARCHAR Observacao
    }

📖 Dicionário das Tabelas e Cardinalidades

🏢 Corporativo & Filiais

  • PK Empresa.ID_Empresa
  • UK Empresa.CNPJ (14 dígitos)
  • PK Filial.ID_Filial
  • FK Filial.ID_Empresa → Empresa
  • UK Filial.CNPJ (14 dígitos)

👥 Recursos Humanos

  • PK Cargo.ID_Cargo
  • PK Funcionario.ID_Funcionario
  • FK Funcionario.ID_Cargo → Cargo
  • FK Funcionario.ID_Filial → Filial
  • UK Funcionario.CPF & E_mail

👤 Clientes & Contatos

  • PK Cliente.ID_Cliente
  • UK Cliente.CPF (Regex: 11 dígitos)
  • UK Cliente.Celular & E_mail
  • PK Endereco.ID_Endereco
  • FK Vinculado exclusivamente (nonnulls)

📦 Catálogo, Estoque & Fornecedor

  • PK Categoria.ID_Categoria
  • PK Estoque.ID_Estoque
  • PK Produto.ID_Produto
  • FK Produto.ID_Categoria & ID_Estoque
  • PK Fornecedor_Produto (N:M)

🛒 Vendas, Pedidos & Itens

  • PK Pedido.ID_Pedido
  • FK Pedido.ID_Cliente & ID_Filial
  • FK Pedido.ID_Funcionario & ID_Pagamento
  • PK Itens_Pedido.ID_Item
  • FK Itens_Pedido.ID_Pedido & ID_Produto

💳 Pagamento & Rastreamento

  • PK Pagamento.ID_Pagamento
  • PK Movimentacao_Estoque.ID_Movimentacao
  • FK Movimentacao → Estoque & Produto
  • CHK Tipo IN ('Entrada', 'Saída')
